// Bake the game's own face painter into the far-crowd texture atlas sources.
import { chromium } from '@playwright/test';
import { writeFile } from 'node:fs/promises';
const browser=await chromium.launch({headless:true});
process.on('SIGTERM',()=>{void browser.close().finally(()=>process.exit(143));});
try {
 const page=await browser.newPage();
 page.on('pageerror',e=>console.error(e.message));
 await page.goto('http://127.0.0.1:5217/tmp/crowd-study/paint.html');
 await page.waitForFunction(()=>document.body.dataset.ready==='true');
 const faces=await page.evaluate(()=>window.paint());
 for(const [id,data] of Object.entries(faces)) await writeFile(`tmp/crowd-study/${id}.png`,Buffer.from(data,'base64'));
 console.log('Painted',Object.keys(faces).length,'crowd source textures');
} finally {await browser.close();}
